```lua
local M = {}
function M.new_reg() return { cells = {}, next_id = 1 } end
function M.qm_bind(reg, name)
  local id = reg.next_id; reg.next_id = id + 1
  local cell = { id = id, name = name, act = 0, edges = {} }
  reg.cells[id] = cell
  return cell
end
function M.qm_link(reg, src_id, dst_id, weight)
  local dst = reg.cells[dst_id]
  if not dst then return nil end
  for i, e in ipairs(dst.edges) do
    if e.src == src_id then e.weight = weight; return e end
  end
  local edge = { src = src_id, dst = dst_id, weight = weight }
  dst.edges[#dst.edges + 1] = edge
  return edge
end
function M.qm_evict(reg, cell_id, cap)
  local cell = reg.cells[cell_id]
  if not cell then return nil end
  while #cell.edges > cap do
    local ci = 1
    for i = 2, #cell.edges do
      if cell.edges[i].weight < cell.edges[ci].weight then ci = i end
    end
    table.remove(cell.edges, ci)
  end
  return cell
end
function M.view_wsum(reg, cell_id)
  local cell = reg.cells[cell_id]
  if not cell then return 0 end
  local sum = 0
  for _, e in ipairs(cell.edges) do sum = sum + e.weight end
  return sum
end
function M.clamp01(v)
  if v > 1 then return 1 end
  if v < 0 then return 0 end
  return v
end
return M
```
