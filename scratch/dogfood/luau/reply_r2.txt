local M = {}

function M.track(side, paddleY, ballY)
    local speed = side == "left" and 0.85 or 0.70
    local dead = 1.5
    local y = paddleY
    local d = ballY - y
    if math.abs(d) > dead then
        local step = math.min(speed, math.abs(d) - dead)
        y = y + (d > 0 and 1 or -1) * step
    end
    y = math.max(6, math.min(54, y))
    return y
end

return M
