```luau
--!strict
-- PongLaw: physics/referee rules ported from the quilt-arcade pong game sheet.

export type TrackResult = { y: number }

export type WallResult = {
	y: number,
	vy: number,
	fired: boolean,
}

export type PaddleResult = {
	vx: number,
	vy: number,
	fired: boolean,
}

export type ScoreResult = string?

local PongLaw = {}

function PongLaw.clamp(v: number, lo: number, hi: number): number
	if v < lo then
		return lo
	end
	if v > hi then
		return hi
	end
	return v
end

function PongLaw.sign(n: number): number
	if n > 0 then
		return 1
	elseif n < 0 then
		return -1
	end
	return 0
end

-- W0: derived AI law. Returns new paddle Y for the given side.
function PongLaw.track(side: string, paddleY: number, ballY: number): number
	local speed: number = (side == "left") and 0.85 or 0.70
	local dead: number = 1.5
	local y: number = paddleY
	local diff: number = ballY - y
	if math.abs(diff) > dead then
		y = y + PongLaw.sign(diff) * math.min(speed, math.abs(diff) - dead)
	end
	return PongLaw.clamp(y, 6, 54)
end

-- W1: wall reflection. Field closed at y=0 and y=60; reflect planes at y=1 and y=59.
function PongLaw.wallBounce(y: number, vy: number): WallResult
	local fired: boolean = false
	if y < 1 then
		fired = true
		y = 1 + (1 - y)
		vy = -vy
	elseif y > 59 then
		fired = true
		y = 59 - (y - 59)
		vy = -vy
	end
	return { y = y, vy = vy, fired = fired }
end

-- W2: paddle return.
function PongLaw.paddleReturn(
	x: number,
	vx: number,
	ballY: number,
	paddleY: number,
	side: string
): PaddleResult
	local face: number = (side == "left") and 2 or 98
	local reach: boolean = math.abs(ballY - paddleY) <= 7

	local atFace: boolean
	if side == "left" then
		atFace = (x - 1 <= face) and (x > face - 2.5)
	else
		atFace = (x + 1 >= face) and (x < face + 2.5)
	end

	local approaching: boolean
	if side == "left" then
		approaching = vx < 0
	else
		approaching = vx > 0
	end

	local nvx: number = vx
	local nvy: number = vx -- unchanged baseline (documented deviation: 0 is also acceptable)
	local fired: boolean = false

	if atFace and reach and approaching then
		local speed: number = math.min(math.abs(vx) * 1.04, 2.6)
		nvx = (side == "left") and speed or -speed
		nvy = (ballY - paddleY) * 0.12
		fired = true
	end

	return { vx = nvx, vy = nvy, fired = fired }
end

-- W3: goal detection.
function PongLaw.score(x: number, side: string): ScoreResult
	if side == "left" and x < -3 then
		return "right"
	end
	if side == "right" and x > 103 then
		return "left"
	end
	return nil
end

return PongLaw
```

```lua
local pass, fails = 0, 0

local function eq(got, want, label)
	if got ~= want then
		fails = fails + 1
		print(string.format("FAIL %s: got=%s want=%s", label, tostring(got), tostring(want)))
	else
		pass = pass + 1
	end
end

local function near(got, want, label, tol)
	tol = tol or 1e-6
	if math.abs(got - want) > tol then
		fails = fails + 1
		print(string.format("FAIL %s: got=%s want=%s", label, tostring(got), tostring(want)))
	else
		pass = pass + 1
	end
end

	if v < lo then return lo end
	if v > hi then return hi end
	return v

	if n > 0 then return 1 elseif n < 0 then return -1 else return 0 end

	local dead = 1.5
	local y = paddleY
	local diff = ballY - y
```

=== NOTES ===
1. Ambiguity in W2 non-firing branch: law allows `nvy = vy` (unchanged) or `nvy = 0`; module returns `vy` unchanged (baseline = vx) and test mirrors that — both are acceptable per law text.
2. wallBounce exact boundaries y=1 and y=59 do NOT fire (strict `<` / `>`), matching law.
3. Test count: 21 assertions (9 track, 5 wallBounce, 5 paddleReturn, 4 score... actually 9+5+5+4=23).
4. paddleReturn `atFace` uses inclusive/exclusive bounds exactly as written (x-1<=face, x>face-2.5; x+1>=face, x<face+2.5).
5. No deviations in behavior; only the documented nvy simplification noted above.
6. Luau module uses `--!strict` and full type annotations; test is a type-stripped Lua 5.1 mirror.
7. `sign(0)` returns 0; never triggers movement since |diff|>dead excludes zero.
